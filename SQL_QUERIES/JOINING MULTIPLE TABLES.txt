
************************************************************ ASSIGNMENT ON JOINING MULTIPLE TABLES ******************************************************************


Q1) WAQTD ENAME, MANAGER'S NAME AND EMP LOC FOR THE EMP WORKING AS CLERK
--->
ORACLE
=======
SELECT E1.ENAME, E2.ENAME AS MANAGER, D1.LOC
FROM EMP E1, EMP E2, DEPT D1
WHERE E1.MGR=E2.EMPNO AND E1.DEPTNO=D1.DEPTNO AND E1.JOB='CLERK';

ANSI
=====
SELECT E1.ENAME, E2.ENAME AS MANAGER, D1.LOC
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO
WHERE E1.JOB='CLERK';


Q2) WAQTD ENAME, MANAGER'S SALARY AND MANAGER'S LOCATION IF MANAGER WORKING AS PRESIDENT.
--->
ORACLE
=======
SELECT E1.ENAME, E2.SAL, D2.LOC
FROM EMP E1, EMP E2, DEPT D2
WHERE E1.MGR=E2.EMPNO AND E2.DEPTNO=D2.DEPTNO AND E2.JOB='PRESIDENT';

ANSI
=====
SELECT E1.ENAME, E2.SAL, D2.LOC
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D2 ON E2.DEPTNO=D2.DEPTNO
WHERE E2.JOB='PRESIDENT';


Q3) WAQTD ENAME, EMP SAL , MANAGER'S NAME, MANAGER'S SAL ALONG WITH EMP DNAME IF EMP EARNS MORE THAN MANAGER.
--->
ORACLE
=======
SELECT E1.ENAME, E1.SAL, E2.ENAME, E2.SAL, D1.DNAME
FROM EMP E1, EMP E2, DEPT D1
WHERE E1.MGR=E2.EMPNO AND E1.DEPTNO=D1.DEPTNO AND E1.SAL>E2.SAL;

ANSI
=====
SELECT E1.ENAME, E1.SAL, E2.ENAME, E2.SAL, D1.DNAME
FROM EMP E1 
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO
WHERE E1.SAL>E2.SAL;


Q4) WAQTD ENAME, SAL, MANAGER'S NAME, MANAGER'S DNAME IF MANAGER WORK IN NEW YORK WITH SALARY MORE THAN 3000.
--->
ORACLE
=======
SELECT E1.ENAME, E1.SAL, E2.ENAME, D2.DNAME
FROM EMP E1, EMP E2, DEPT D2
WHERE E1.MGR=E2.EMPNO AND E2.DEPTNO=D2.DEPTNO AND D2.LOC='NEW YORK' AND E2.SAL>3000;

ANSI
=====
SELECT E1.ENAME, E1.SAL, E2.ENAME, D2.DNAME
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D2 ON E2.DEPTNO=D2.DEPTNO
WHERE D2.LOC='NEW YORK' AND E2.SAL>3000;


Q5) WAQTD ENAME, EMP DNAME, MANAGER'S NAME, MANAGER'S DNAME IF EMP AND MANAGER WORKS IN SAME LOC.
--->
ORACLE
======
SELECT E1.ENAME, D1.DNAME, E2.ENAME, D2.DNAME
FROM EMP E1, EMP E2, DEPT D1, DEPT D2
WHERE E1.MGR=E2.EMPNO AND E1.DEPTNO=D1.DEPTNO AND E2.DEPTNO=D2.DEPTNO AND D1.LOC=D2.LOC;

ANSI
=====
SELECT E1.ENAME, D1.DNAME, E2.ENAME, D2.DNAME
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO
INNER JOIN DEPT D2 ON E2.DEPTNO=D2.DEPTNO
WHERE D1.LOC=D2.LOC;


Q6) WAQTD ENAME, EMP HIREDATE,  MANAGER'S HIREDATE IF MANAGER WAS HIRED BEFORE THE EMP INTO ACCOUTING DEPARTMENT.
--->
ORACLE
======
SELECT E1.ENAME, E1.HIREDATE, E2.HIREDATE
FROM EMP E1, EMP E2, DEPT D1
WHERE E1.MGR=E2.EMPNO AND E1.DEPTNO=D1.DEPTNO AND E1.HIREDATE<E2.HIREDATE AND D1.DNAME='ACCOUNTING';

ANSI
=====
SELECT E1.ENAME, E1.HIREDATE, E2.HIREDATE
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO 
WHERE E1.HIREDATE<E2.HIREDATE AND D1.DNAME='ACCOUNTING';


Q7) WAQTD EMP NAME, MANAGER'S NAME ALONG WITH MANAGER'S MANAGER NAME.
--->
ORACLE
========
SELECT E1.ENAME, E2.ENAME, E3.ENAME
FROM EMP E1, EMP E2, EMP E3
WHERE E1.MGR=E2.EMPNO AND E2.MGR=E3.EMPNO;

ANSI
=====
SELECT E1.ENAME, E2.ENAME, E3.ENAME
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
JOIN EMP E3 ON E2.MGR=E3.EMPNO;


Q8) WAQTD ENAME, DNAME, MANAGER'S NAME HIS DEPT NAME AND MANAGER IS HIRED BEFORE 1982.
--->
ORACLE
=======
SELECT E1.ENAME, D1.DNAME, E2.ENAME, D2.DNAME
FROM EMP E1, EMP E2 , DEPT D1, DEPT D2
WHERE E1.MGR= E2.EMPNO  AND E1.DEPTNO=D1.DEPTNO AND E2.DEPTNO=D2.DEPTNO AND E2.HIREDATE<'01-JAN-1982';

ANSI
=====
SELECT E1.ENAME, D1.DNAME, E2.ENAME, D2.DNAME
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO
INNER JOIN DEPT D2 ON E2.DEPTNO=D2.DEPTNO
WHERE E2.HIREDATE<'01-JAN-1982';


Q9) WAQTD EMP NAME WITH DNAME , MANAGER'S NAME WITH LOCATION ALONG WITH MANAGER'S MANAGER NAME WITH LOC.
--->
ORACLE
========
SELECT E1.ENAME, D1.DNAME, E2.ENAME, D2.LOC, E3.ENAME, D3.LOC
FROM EMP E1, EMP E2, EMP E3, DEPT D1, DEPT D2, DEPT D3
WHERE E1.MGR=E2.EMPNO AND E2.MGR=E3.EMPNO AND E1.DEPTNO=D1.DEPTNO AND E2.DEPTNO=D2.DEPTNO AND E3.DEPTNO=D3.DEPTNO;

ANSI
=====
SELECT E1.ENAME, D1.DNAME, E2.ENAME, D2.LOC, E3.ENAME, D3.LOC
FROM EMP E1
JOIN EMP E2 ON E1.MGR=E2.EMPNO
JOIN EMP E3 ON E2.MGR=E3.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO
INNER JOIN DEPT D2 ON E2.DEPTNO=D2.DEPTNO
INNER JOIN DEPT D3 ON E3.DEPTNO=D3.DEPTNO;


Q10) WAQTD EMP NAME WITH DNAME, MANAGER'S SAL WITH DNAME ALONG WITH MANAGER'S MANAGER JOB WITH DNAME IF EMP WORK IN DEPT NO 10 , MANAGER WORK AS PRESIDENT OR ACTUAL MANAGER AND MANAGER'S MANAGER SALARY MUST BE GREATER OR EQUAL TO MANAGER'S SALARY.
--->
ORACLE
======
SELECT E1.ENAME, D1.DNAME, E2.SAL, D2.DNAME, E3.JOB,D3.DNAME
FROM EMP E1, EMP E2 , DEPT D1, DEPT D2, EMP E3, DEPT D3
WHERE E1.MGR=E2.EMPNO AND E2.MGR=E3.EMPNO AND E1.DEPTNO=D1.DEPTNO AND E2.DEPTNO=D2.DEPTNO AND E3.DEPTNO=D3.DEPTNO AND E1.DEPTNO=10 AND E2.JOB IN ('PRESIDENT','MANAGER') AND E3.SAL>=E2.SAL;

ANSI
=====
SELECT E1.ENAME, D1.DNAME, E2.SAL, D2.DNAME, E3.JOB,D3.DNAME
FROM EMP E1 
JOIN EMP E2 ON E1.MGR=E2.EMPNO
JOIN EMP E3 ON E2.MGR=E3.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO=D1.DEPTNO
INNER JOIN DEPT D2 ON E2.DEPTNO=D2.DEPTNO
INNER JOIN DEPT D3 ON E3.DEPTNO=D3.DEPTNO
WHERE E1.DEPTNO=10 AND E2.JOB IN ('PRESIDENT','MANAGER') AND E3.SAL>=E2.SAL;
