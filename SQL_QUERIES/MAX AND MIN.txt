
************************************************************ ASSIGNMENT ON MIN AND MAX **********************************************************************

Q41) WAQTD NAME OF THE EMPLOYEE EARNING MAXIMUM SALARY.
--->
SELECT ENAME 
FROM EMP
WHERE SAL = (SELECT MAX(SAL)
             FROM EMP
             );

Q42) WAQTD NAME OF EMPLOYEE EARNING MINIMUM SALARY.
--->
SELECT ENAME
FROM EMP
WHERE SAL = (SELECT MIN(SAL)
             FROM EMP
             );

Q43) WAQTD NAME AND HIREDATE OF THE EMPLOYEE HIRED BEFORE ALL THE EMPLOYEES (FIRST EMP).
--->
SELECT ENAME, HIREDATE
FROM EMP
WHERE HIREDATE = (SELECT MIN(HIREDATE)
                  FROM EMP 
                  );
