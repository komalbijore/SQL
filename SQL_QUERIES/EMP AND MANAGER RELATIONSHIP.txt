
****************************************************** ASSIGNMENT ON EMP AND MANAGER RELATION *************************************************************

Q71) WAQTD SMITH'S REPORTING MANAGER'S NAME.
--->
SELECT *
FROM EMP
WHERE EMPNO = (SELECT MGR
               FROM EMP
               WHERE ENAME= 'SMITH'
               );

Q72) WAQTD ADAMS MANAGER'S MANAGER NAME.
--->
SELECT ENAME
FROM EMP
WHERE EMPNO = ( SELECT MGR
                 FROM EMP
                 WHERE EMPNO = (SELECT MGR
                                FROM EMP
                                WHERE ENAME='ADAMS'
                               )
                );

Q73) WAQTD DNAME OF JONES MANAGER.
--->
